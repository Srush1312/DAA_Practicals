import heapq

class Node:
    def __init__(self, char, freq):
        self.char = char
        self.freq = freq
        self.left = None
        self.right = None

    def __lt__(self, other):
        return self.freq < other.freq


def huffman_encoding(chars, freq):
    heap = []

    for char, frequency in zip(chars, freq):
        heapq.heappush(heap, Node(char, frequency))

    while len(heap) > 1:
        left = heapq.heappop(heap)
        right = heapq.heappop(heap)

        new_node = Node(None, left.freq + right.freq)
        new_node.left = left
        new_node.right = right

        heapq.heappush(heap, new_node)

    codes = {}

    def generate_codes(root, code):
        if root is None:
            return

        if root.char is not None:
            codes[root.char] = code if code else "0"
            return

        generate_codes(root.left, code + "0")
        generate_codes(root.right, code + "1")

    generate_codes(heap[0], "")

    print("Huffman Codes:")
    for char in chars:
        print(char, ":", codes[char])


n = int(input("Enter number of characters: "))

chars = []
freq = []

for i in range(n):
    ch = input("Enter character: ")
    f = int(input("Enter frequency: "))
    chars.append(ch)
    freq.append(f)

huffman_encoding(chars, freq)
