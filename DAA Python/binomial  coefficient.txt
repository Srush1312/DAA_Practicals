def binomial_coefficient(n):
    dp = [[0 for j in range(n + 1)] for i in range(n + 1)]

    for i in range(n + 1):
        for j in range(i + 1):
            if j == 0 or j == i:
                dp[i][j] = 1
            else:
                dp[i][j] = dp[i - 1][j - 1] + dp[i - 1][j]

    print("Pascal's Triangle:")

    for i in range(n + 1):
        for j in range(i + 1):
            print(dp[i][j], end=" ")
        print()

    return dp


n = int(input("Enter the value of n: "))

dp = binomial_coefficient(n)

r = int(input("Enter the value of r: "))

if r >= 0 and r <= n:
    print("Binomial Coefficient C(", n, ",", r, ") =", dp[n][r])
else:
    print("Invalid value of r")
